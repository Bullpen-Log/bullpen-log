#!/bin/zsh
# 공개 시험대를 끈다 — 이 시험대가 띄운 것만(3101 · 54398 · 54399). 3000 은 건드리지 않는다
cd "$(dirname "$0")"
for f in harness.pid dev.pid; do [ -f $f ] && kill $(cat $f) 2>/dev/null; done
for port in 3101 54398 54399; do
  for pid in $(lsof -tiTCP:$port -sTCP:LISTEN 2>/dev/null); do kill $pid 2>/dev/null; done
done
sleep 2
lsof -iTCP -sTCP:LISTEN -nP | grep -E ':(3101|54398|54399)\b' || echo 'ports free'
