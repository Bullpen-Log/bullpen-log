async function page({ w = 390, h = 844, ua = UA_IPHONE, mobile = true } = {}) {
  const { browserContextId } = await send('Target.createBrowserContext');
  const { targetId } = await send('Target.createTarget', { url: 'about:blank', browserContextId });
  const { sessionId } = await send('Target.attachToTarget', { targetId, flatten: true });
  const s = (m, p) => send(m, p, sessionId);
  await s('Page.enable'); await s('Runtime.enable'); await s('DOM.enable'); await s('Network.enable');
  await s('Emulation.setDeviceMetricsOverride', { width: w, height: h, deviceScaleFactor: 2, mobile });
  await s('Emulation.setUserAgentOverride', { userAgent: ua });
  const errors = [];
  listeners.add((m) => { if (m.sessionId === sessionId && m.method === 'Page.javascriptDialogOpening') void s('Page.handleJavaScriptDialog', { accept: true }); });
  listeners.add((m) => { if (m.sessionId === sessionId && m.method === 'Runtime.exceptionThrown') errors.push(m.params.exceptionDetails.text); });
  const ev = async (expr) => { const r = await s('Runtime.evaluate', { expression: expr, awaitPromise: true, returnByValue: true }); if (r.exceptionDetails) throw new Error(r.exceptionDetails.text + expr); return r.result.value; };
  const text = () => ev('document.body.innerText');
  const waitText = async (t, ms = 30000) => { const end = Date.now() + ms; for (;;) { const b = await text().catch(() => ''); if (b.includes(t)) return true; if (Date.now() > end) throw new Error(`못 찾음: ${t}\n--- 화면\n${b.slice(0, 500)}`); await sleep(120); } };
  const click = async (t, ms = 30000) => { const end = Date.now() + ms; const q = JSON.stringify(t); for (;;) { const done = await ev(`(()=>{const c=[...document.querySelectorAll('button,a,label,[role=button]')].filter(e=>e.innerText.trim().includes(${q})&&!e.disabled&&e.getAttribute('aria-disabled')!=='true'&&e.offsetParent!==null);const e=c.at(-1);if(!e)return false;e.click();return true})()`); if (done) return; if (Date.now() > end) throw new Error('못 누름: ' + t + '\n' + (await text()).slice(0, 500)); await sleep(120); } };
  const type = async (sel, v) => { await ev(`(()=>{const e=document.querySelector(${JSON.stringify(sel)});e.focus();e.select&&e.select();})()`); await s('Input.insertText', { text: v }); };
  const files = async (paths, nth = 0) => { const { root } = await s('DOM.getDocument', { depth: 1 }); const { nodeIds } = await s('DOM.querySelectorAll', { nodeId: root.nodeId, selector: 'input[type=file]' }); await s('DOM.setFileInputFiles', { nodeId: nodeIds[nth], files: paths }); };
  const go = async (url) => { await s('Page.navigate', { url }); await sleep(400); };
  const overflow = () => ev('document.documentElement.scrollWidth - document.documentElement.clientWidth');
  const offline = (on) => s('Network.emulateNetworkConditions', { offline: on, latency: 0, downloadThroughput: -1, uploadThroughput: -1 });
  return { s, ev, text, waitText, click, type, files, go, overflow, errors, offline, close: () => send('Target.disposeBrowserContext', { browserContextId }) };
}
