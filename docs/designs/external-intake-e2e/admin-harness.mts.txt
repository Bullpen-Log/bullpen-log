function sign(url: string, ttl: number) {
  const b = (v: object) => Buffer.from(JSON.stringify(v)).toString('base64url');
  const head = `${b({ alg: 'HS256' })}.${b({ url, exp: Math.floor(Date.now() / 1000) + ttl })}`;
  return `${head}.${createHmac('sha256', SECRET).update(head).digest('base64url')}`;
}
function verify(token: unknown, name: string) {
  if (typeof token !== 'string') return 'Missing signature';
  const [h, p, s] = token.split('.');
  if (!s || createHmac('sha256', SECRET).update(`${h}.${p}`).digest('base64url') !== s) return 'invalid signature';
  const pl = JSON.parse(Buffer.from(p, 'base64url').toString());
  if (pl.url !== `${INTAKE_BUCKET}/${name}`) return 'invalid signature';
  if (pl.exp * 1000 < Date.now()) return 'jwt expired signature';
  return null;
}
const send = (res: ServerResponse, status: number, body: unknown = null, headers: Record<string, string> = {}) => {
  res.writeHead(status, { ...CORS, 'Tus-Resumable': '1.0.0', 'Content-Type': 'application/json', ...headers });
  res.end(body == null ? undefined : JSON.stringify(body, (_, v) => (typeof v === 'bigint' ? Number(v) : v)));
};
const bodyOf = (req: IncomingMessage) => new Promise<Buffer>((r) => { const c: Buffer[] = []; req.on('data', (x) => c.push(x)); req.on('end', () => r(Buffer.concat(c))); });
const meta = (h: unknown) => Object.fromEntries(String(h ?? '').split(',').map((x) => x.trim().split(' ')).map(([k, v]) => [k, Buffer.from(v ?? '', 'base64').toString()]));

async function tus(req: IncomingMessage, res: ServerResponse, id: string | null) {
  if (req.method === 'POST') {
    const m = meta(req.headers['upload-metadata']);
    const why = verify(req.headers['x-signature'], m.objectName);
    if (why) return send(res, 400, { message: why });
    if (st.objects.has(m.objectName)) return send(res, 409, { message: 'Asset Already Exists' });
    const uid = randomBytes(12).toString('hex');
    const file = path.join(dir, `tus-${uid}`);
    writeFileSync(file, '');
    st.uploads.set(uid, { object: m.objectName, length: Number(req.headers['upload-length']), offset: 0, file });
    tlog({ m: 'POST', uid, object: m.objectName });
    return send(res, 201, null, { Location: `http://127.0.0.1:${ST_PORT}/storage/v1/upload/resumable/sign/${uid}` });
  }
  const up = id ? st.uploads.get(id) : null;
  if (!up) return send(res, 404, { message: 'Upload not found' });
  const why = verify(req.headers['x-signature'], up.object);
  if (why) return send(res, 400, { message: why });
  if (req.method === 'HEAD') return send(res, 200, null, { 'Upload-Offset': String(up.offset), 'Upload-Length': String(up.length), 'Cache-Control': 'no-store' });
  if (req.method === 'PATCH') {
    if (Number(req.headers['upload-offset']) !== up.offset) return send(res, 409, { message: 'offset conflict' });
    const data = await bodyOf(req);
    const final = up.offset + data.length >= up.length;
    if (!final && data.length !== TUS_CHUNK_BYTES) return send(res, 400, { message: 'chunk must be 6MB' });
    appendFileSync(up.file, data);
    up.offset += data.length;
    tlog({ m: 'PATCH', object: up.object, offset: up.offset - data.length, len: data.length });
    if (up.offset === up.length) st.objects.set(up.object, up.file);
    return send(res, 204, null, { 'Upload-Offset': String(up.offset) });
  }
  send(res, 405);
}

async function rest(req: IncomingMessage, res: ServerResponse, url: URL) {
  if (req.method === 'DELETE' && /^\/storage\/v1\/object\/[^/]+\/?$/.test(url.pathname)) {
    const { prefixes = [] } = JSON.parse((await bodyOf(req)).toString() || '{}') as { prefixes?: string[] };
    const gone = prefixes.filter((p) => st.objects.delete(p));
    tlog({ m: 'DELETE', gone });
    return send(res, 200, gone.map((name) => ({ name })));
  }
  const ls = /^\/storage\/v1\/object\/list\/([^/]+)$/.exec(url.pathname);
  if (ls && req.method === 'POST') {
    const { prefix = '' } = JSON.parse((await bodyOf(req)).toString() || '{}') as { prefix?: string };
    const names = [...st.objects.keys()].filter((k) => k.startsWith(`${prefix}/`)).map((k) => k.slice(prefix.length + 1)).filter((k) => !k.includes('/'));
    return send(res, 200, names.map((name) => ({ name, id: name, metadata: {} })));
  }
  const su = /^\/storage\/v1\/object\/upload\/sign\/([^/]+)\/(.+)$/.exec(url.pathname);
  if (su && req.method === 'POST') {
    await bodyOf(req);
    const name = decodeURIComponent(su[2]);
    const upsert = String(req.headers['x-upsert'] ?? '') === 'true';
    if (st.objects.has(name) && !upsert) return send(res, 400, { statusCode: '409', error: 'Duplicate', message: 'The resource already exists' });
    const token = sign(`${su[1]}/${name}`, 7200);
    return send(res, 200, { url: `/object/upload/sign/${su[1]}/${name}?token=${token}`, token });
  }
  /* 서명 올리기 주소로 PUT — 분석 서버가 결과 · 미리보기를 올린다 */
  if (su && req.method === 'PUT') {
    const name = decodeURIComponent(su[2]);
    const why = verify(url.searchParams.get('token'), name);
    if (why) return send(res, 400, { message: why });
    putObject(name, await bodyOf(req));
    tlog({ m: 'PUT', object: name });
    return send(res, 200, { Key: `${su[1]}/${name}` });
  }
  const sr = /^\/storage\/v1\/object\/sign\/([^/]+)\/(.+)$/.exec(url.pathname);
  if (sr && req.method === 'POST') {
    await bodyOf(req);
    const name = decodeURIComponent(sr[2]);
    if (!st.objects.has(name)) return send(res, 400, { statusCode: '404', error: 'not_found', message: 'Object not found' });
    return send(res, 200, { signedURL: `/object/sign/${sr[1]}/${name}?token=${sign(`${sr[1]}/${name}`, 300)}` });
  }
  if (sr && req.method === 'GET') {
    const name = decodeURIComponent(sr[2]);
    const why = verify(url.searchParams.get('token'), name);
    if (why) return send(res, 400, { statusCode: '403', message: why });
    const file = st.objects.get(name);
    if (!file) return send(res, 400, { statusCode: '404', message: 'Object not found' });
    tlog({ m: 'GET', object: name, range: req.headers.range ?? null, download: url.searchParams.get('download') });
    const size = statSync(file).size;
    const range = /^bytes=(\d+)-(\d+)?$/.exec(req.headers.range ?? '');
    const disp: Record<string, string> = url.searchParams.has('download') ? { 'Content-Disposition': `attachment; filename*=UTF-8''${encodeURIComponent(url.searchParams.get('download') || name.split('/').pop()!)}` } : {};
    if (range) {
      const start = Number(range[1]); const end = Math.min(size - 1, Number(range[2] ?? size - 1));
      res.writeHead(206, { ...CORS, 'Content-Type': 'video/quicktime', 'Content-Range': `bytes ${start}-${end}/${size}`, 'Content-Length': String(end - start + 1), ...disp });
      return void createReadStream(file, { start, end }).pipe(res);
    }
    res.writeHead(200, { ...CORS, 'Content-Type': 'video/quicktime', 'Content-Length': String(size), ...disp });
    return void createReadStream(file).pipe(res);
  }
  send(res, 404, { message: 'no route' });
}

/* 가짜 GPU — 약속(lib/pitch-3d/v2/gpu-client.ts): POST /jobs → { callId }, GET /jobs/:id → { status }, GET /engine → { build } */
async function gpu(req: IncomingMessage, res: ServerResponse, url: URL) {
  if (req.headers['modal-key'] !== GPU_KEY || req.headers['modal-secret'] !== GPU_SECRET) return send(res, 401, { message: 'auth' });
  if (url.pathname === '/gpu/engine') return send(res, 200, { build: GPU_BUILD });
  if (url.pathname === '/gpu/jobs' && req.method === 'POST') {
    const body = JSON.parse((await bodyOf(req)).toString());
    const callId = `fc-${randomBytes(6).toString('hex')}`;
    const job = { status: 'running', req: { jobId: body.jobId, meta: body.meta, hasPreviews: !!body.previews }, fetched: {} as Record<string, number> };
    st.gpu.set(callId, job);
    void (async () => {
      /* 받은 서명 주소로 원본을 내려받는다 — 크기만 적는다 */
      for (const [k, u] of [['side', body.side.url], ['back', body.back.url]] as const) {
        const r = await fetch(u); job.fetched[k] = (await r.arrayBuffer()).byteLength;
      }
      const sc = base; const { side: cs, back: cb } = cameras(sc);
      const ts = makeV2Track(sc, cs, sc.side, 'side', 11); const tb = makeV2Track(sc, cb, sc.back, 'back', 12);
      const r = fitPitch3dV2({ side: ts.track, back: tb.track, hand: sc.hand, heightCm: 183, jobId: body.jobId, poseModel: 'synth', screenRecorded: true, slowmoFps: 240, events: { kneeUp: ts.toMedia(EV.kneeUp), footPlant: ts.toMedia(EV.footPlant), release: ts.toMedia(EV.release) } }).result;
      const put = (u: string, b: Buffer) => fetch(u, { method: 'PUT', body: b, headers: { 'content-type': 'application/octet-stream' } }).then((x) => x.status);
      job.fetched.result = await put(body.result.uploadUrl, Buffer.from(JSON.stringify(r)));
      if (body.previews) { await put(body.previews.side, mov240(4096)); await put(body.previews.back, mov120(4096)); }
      job.status = 'done';
    })().catch((e) => { console.error('gpu', e); job.status = 'failed'; });
    return send(res, 200, { callId });
  }
  const j = /^\/gpu\/jobs\/([^/]+)$/.exec(url.pathname);
  if (j) { const job = st.gpu.get(j[1]); return job ? send(res, 200, { status: job.status, stage: job.status === 'running' ? 'fit' : undefined, stages: {} }) : send(res, 404, {}); }
  send(res, 404, {});
}

async function control(req: IncomingMessage, res: ServerResponse, url: URL) {
  const name = url.pathname.split('/')[2];
  const q = url.searchParams;
  if (name === 'fixture') { const b = ({ side: mov240(20 * 1024 * 1024 + 123), rear: mov120(14 * 1024 * 1024 + 77) } as Record<string, Buffer>)[q.get('f')!]; res.writeHead(200, { 'Content-Length': String(b.length) }); return void res.end(b); }
  /* 읽기만 하는 SQL — 격리 PGlite A 뿐 */
  if (name === 'sql') { const sqlText = q.get('q')!; if (!/^\s*select\b/i.test(sqlText)) return send(res, 400, { message: 'select only' }); return send(res, 200, await db.$queryRawUnsafe(sqlText)); }
  if (name === 'seedsha') return send(res, 200, { side: sha(seedBytes.side), rear: sha(seedBytes.rear), s1 });
  if (name === 'objects') return send(res, 200, [...st.objects].map(([k, f]) => ({ k, size: statSync(f).size, sha: sha(readFileSync(f)) })).filter((o) => o.k.includes(q.get('sub') ?? '')));
  if (name === 'gpulog') return send(res, 200, [...st.gpu].map(([id, j]) => ({ id, status: j.status, req: j.req, fetched: j.fetched })));
  if (name === 'log') return send(res, 200, st.log);
  if (name === 'fx') return send(res, 200, { side: sha(mov240(20 * 1024 * 1024 + 123)), rear: sha(mov120(14 * 1024 * 1024 + 77)) });
  /* 같은 접수의 성공한 실행을 하나 더 — 공개 경합에 쓴다(결과는 GPU 와 같은 합성) */
  if (name === 'run') {
    const sc = base; const { side: cs, back: cb } = cameras(sc);
    const ts = makeV2Track(sc, cs, sc.side, 'side', 21); const tb = makeV2Track(sc, cb, sc.back, 'back', 22);
    const id = crypto.randomUUID();
    const s = await db.intakeSubmission.findUniqueOrThrow({ where: { id: q.get('id')! } });
    const revRow = await db.intakeRevision.findUniqueOrThrow({ where: { id: s.currentRevisionId! } });
    const r = fitPitch3dV2({ side: ts.track, back: tb.track, hand: sc.hand, heightCm: 183, jobId: id, poseModel: 'synth', screenRecorded: true, slowmoFps: 240, events: { kneeUp: ts.toMedia(EV.kneeUp), footPlant: ts.toMedia(EV.footPlant), release: ts.toMedia(EV.release) } }).result;
    const raw = Buffer.from(JSON.stringify(r)); const key = runResultPath(s.id, id); putObject(key, raw);
    await db.intakeAnalysisRun.create({ data: { id, submissionId: s.id, revisionId: revRow.id, engineKey: 'e2e-second', status: 'succeeded', resultPath: key, resultMeta: { input: { side: revRow.sideAssetId, rear: revRow.rearAssetId, hand: 'R' }, sha256: sha(raw) }, finishedAt: new Date() } });
    return send(res, 200, { runId: id });
  }
  send(res, 404, {});
}
