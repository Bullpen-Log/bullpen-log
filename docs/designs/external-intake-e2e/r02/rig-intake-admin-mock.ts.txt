import { CURRENT_REVISION_ID, RUNS, SID, fixtureView, rigState } from '@/app/r02-rig/state';

export type IntakeActionFailure = { ok: false; code: string; message: string; detail?: Record<string, unknown> };
export type IntakeActionResult<T extends object = object> = ({ ok: true } & T) | IntakeActionFailure;

const log = (name: string, args: unknown[]) => void rigState().calls.push({ name, args, at: Date.now() });
const fail = (code: string, message: string): IntakeActionFailure => ({ ok: false, code, message });
const stub = async (name: string, args: unknown[]): Promise<IntakeActionResult> => (log(name, args), fail('rig_stub', '시험대에서 쓰지 않는 동작이에요.'));

export async function getIntakeRunReview(submissionId: string, runId: string) {
  log('getIntakeRunReview', [submissionId, runId]);
  const s = rigState();
  if (s.reviewDelayMs) await new Promise((r) => setTimeout(r, s.reviewDelayMs));
  const run = RUNS.find((r) => r.id === runId);
  if (submissionId !== SID || !run) return fail('not_found', '실행을 찾지 못했어요.');
  return {
    ok: true as const,
    view: fixtureView,
    athleteName: '합성 투수',
    revisionNumber: run.revisionNumber,
    isCurrentRevision: run.isCurrentRevision,
    finishedAt: null,
    resultDigest: run.digest,
    currentRevisionId: CURRENT_REVISION_ID,
  };
}

export async function acknowledgeIntakeRunReview(submissionId: string, runId: string, resultDigest: string, currentRevisionId: string | null) {
  log('acknowledgeIntakeRunReview', [submissionId, runId, resultDigest, currentRevisionId]);
  const s = rigState();
  if (s.ackRefuse) {
    const code = s.ackRefuse;
    s.ackRefuse = null;
    return fail(code, code === 'stale_review' ? '검토를 연 뒤 결과나 접수가 바뀌었어요. 다시 열어 확인해 주세요.' : '먼저 3D 결과를 열어 검토해 주세요.');
  }
  const run = RUNS.find((r) => r.id === runId);
  if (!run || run.digest !== resultDigest || currentRevisionId !== CURRENT_REVISION_ID) return fail('stale_review', '검토를 연 뒤 결과나 접수가 바뀌었어요.');
  const id = `ack-${++s.seq}`;
  s.acks[id] = { runId, digest: resultDigest, revisionId: currentRevisionId };
  return { ok: true as const, acknowledgementId: id };
}

export async function publishIntakeRun(
  submissionId: string,
  runId: string,
  explanation: string,
  expectedActivePublicationId: string | null,
  options: { allowOlderRevision?: boolean; acknowledgementId?: string | null } = {},
) {
  log('publishIntakeRun', [submissionId, runId, explanation, expectedActivePublicationId, options]);
  const s = rigState();
  const ack = options?.acknowledgementId ? s.acks[options.acknowledgementId] : undefined;
  if (!ack || ack.runId !== runId) return fail('review_required', '먼저 3D 결과를 열어 검토해 주세요.');
  if (expectedActivePublicationId !== s.activePublicationId) return fail('stale_publication', '그사이 공개가 바뀌었어요.');
  const replaced = s.publishedRunId != null;
  s.publishedRunId = runId;
  s.activePublicationId = `pub-${++s.seq}`;
  return { ok: true as const, replaced };
}
