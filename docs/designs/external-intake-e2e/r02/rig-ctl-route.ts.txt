/* R02 시험대 전용 조작 — 상태 읽기 · 모드 바꾸기 · 처음으로. 루프백에서만 띄운다. 제품 diff 에 없다 */
import { resetRig, rigState } from '../state';

export async function GET() {
  return Response.json(rigState());
}

export async function POST(req: Request) {
  const body = (await req.json()) as { reset?: boolean; ackRefuse?: 'stale_review' | 'review_required' | null; reviewDelayMs?: number };
  if (body.reset) resetRig();
  const s = rigState();
  if (body.ackRefuse !== undefined) s.ackRefuse = body.ackRefuse;
  if (body.reviewDelayMs !== undefined) s.reviewDelayMs = body.reviewDelayMs;
  return Response.json(s);
}
