# 확장 관리 시험대: 가짜 저장소 · 가짜 GPU · PGlite 둘 + next dev 3010(접수 표 있음) · 3101(접수 표 없음). 운영 값 없음
R=/tmp/intake-rig/intake-e2e; cd $R/clone || exit 1
. $R/admin/admin-env.sh
nohup node --import ./scripts/alias-register.mjs scripts/admin-harness.mts > $R/admin/harness.log 2>&1 &

for i in {1..240}; do grep -q 'harness ready' $R/admin/harness.log && break; sleep 0.5; done
